===JS===
const s = document.getElementById("scenes");
s.style.setProperty("--titre", `"${F.titre}", "Poppins", sans-serif`);
const vertical = H > W;
const cx = W / 2, cy = H * (vertical ? 0.42 : 0.47);
const lueur = $("div", `position:absolute;left:${cx - 700*u}px;top:${cy - 700*u}px;width:${1400*u}px;height:${1400*u}px;border-radius:50%;background:radial-gradient(circle, ${C.accent}55 0%, transparent 62%);opacity:0`);
// le cadeau, dessiné en SVG
const taille = 420 * u;
const cadeau = svg(`<svg viewBox="0 0 200 200" width="${taille}" height="${taille}" style="position:absolute;left:${cx - taille/2}px;top:${cy - taille*0.62}px;overflow:visible">
 <defs><linearGradient id="bo" x1="0" y1="0" x2="1" y2="1"><stop offset="0" stop-color="${C.accent}"/><stop offset="1" stop-color="${C.accent2}"/></linearGradient>
 <linearGradient id="ru" x1="0" x2="1"><stop offset="0" stop-color="#fff" stop-opacity=".95"/><stop offset="1" stop-color="#ffe9a8"/></linearGradient>
 <filter id="om"><feDropShadow dx="0" dy="8" stdDeviation="8" flood-opacity=".35"/></filter></defs>
 <g filter="url(#om)"><rect x="30" y="80" width="140" height="110" rx="10" fill="url(#bo)"/>
 <rect x="22" y="62" width="156" height="34" rx="8" fill="url(#bo)" id="couvercle"/>
 <rect x="91" y="62" width="18" height="128" fill="url(#ru)"/><rect x="30" y="122" width="140" height="16" fill="url(#ru)" opacity=".9"/>
 <path d="M100 62 C 70 20, 40 40, 70 62 Z M100 62 C 130 20, 160 40, 130 62 Z" fill="url(#ru)"/></g></svg>`);
gsap.set(cadeau, { scale: 0, transformOrigin: "50% 85%" });
// le compteur
const chiffre = $("div", `position:absolute;left:0;width:${W}px;text-align:center;top:${cy + taille*0.42}px;font-size:${150*u}px;color:${C.texte};font-family:"${F.titre}","Poppins",sans-serif;font-weight:800`, null, "0");
const ans = $("div", `position:absolute;left:0;width:${W}px;text-align:center;top:${cy + taille*0.42 + 165*u}px;font-size:${44*u}px;color:${C.doux};letter-spacing:.3em`, null, LANG === "fr" ? "ANS" : "YEARS");
// le graphique : barres qui montent en même temps
const g = $("div", `position:absolute;${vertical ? `left:${W*0.12}px;width:${W*0.76}px;top:${H*0.78}px;height:${H*0.12}px` : `left:${W*0.66}px;width:${W*0.26}px;top:${H*0.3}px;height:${H*0.42}px`}`);
const nb = 8, barres = [];
for (let i = 0; i < nb; i++) {
  const b = $("div", `position:absolute;bottom:0;left:${i * 100 / nb + 1.5}%;width:${100 / nb - 3}%;height:100%;border-radius:${10*u}px ${10*u}px 0 0;background:linear-gradient(180deg, ${C.accent2}, ${C.accent}88);transform-origin:bottom;transform:scaleY(0)`, g);
  barres.push(b);
}
const o = { v: 0 };
tl.to(lueur, { opacity: 1, duration: 1.2, ease: "power2.out" }, 0.2);
tl.to(cadeau, { scale: 1, duration: 4, ease: "power2.inOut" }, 0.3);
tl.to(o, { v: 40, duration: 4, ease: "power2.inOut", modifiers: { v: (v) => { chiffre.textContent = Math.round(v); return v; } } }, 0.3);
barres.forEach((b, i) => tl.to(b, { scaleY: (i + 1) / nb, duration: 4 * (i + 1) / nb, ease: "power2.inOut" }, 0.3));
tl.fromTo(chiffre, { scale: 1 }, { scale: 1.25, duration: 0.25, ease: "back.out(3)", yoyo: true, repeat: 1 }, 4.3);
tl.to(cadeau.querySelector("#couvercle"), { y: -40, rotation: -14, transformOrigin: "0% 100%", duration: 0.5, ease: "back.out(2)" }, 4.3);
// le feu d'artifice
const couleurs = [C.accent, C.accent2, "#FFD166", "#ffffff", "#EF476F"];
for (let f = 0; f < 4; f++) {
  const fx = W * (0.2 + 0.6 * alea()), fy = H * (0.15 + 0.25 * alea()), t0 = 4.4 + f * 0.45, col = couleurs[f % couleurs.length];
  const fusee = $("div", `position:absolute;left:${fx}px;top:${H}px;width:${6*u}px;height:${40*u}px;border-radius:${3*u}px;background:linear-gradient(${col}, transparent);opacity:0`);
  tl.set(fusee, { opacity: 1 }, t0 - 0.5);
  tl.to(fusee, { y: fy - H, duration: 0.5, ease: "power2.out" }, t0 - 0.5);
  tl.set(fusee, { opacity: 0 }, t0);
  for (let k = 0; k < 36; k++) {
    const a = (k / 36) * Math.PI * 2, r = (180 + 120 * alea()) * u;
    const p = $("div", `position:absolute;left:${fx}px;top:${fy}px;width:${9*u}px;height:${9*u}px;border-radius:50%;background:${col};box-shadow:0 0 ${14*u}px ${col};opacity:0`);
    tl.set(p, { opacity: 1 }, t0);
    tl.to(p, { x: Math.cos(a) * r, y: Math.sin(a) * r + 60 * u, duration: 1.4, ease: "expo.out" }, t0);
    tl.to(p, { opacity: 0, scale: 0.3, duration: 0.8, ease: "power1.in" }, t0 + 0.7);
  }
}
// tout part en confettis
const t1 = 6.6;
tl.to([cadeau, chiffre, ans, g, lueur], { opacity: 0, scale: 0.9, duration: 0.4, ease: "power2.in" }, t1);
for (let k = 0; k < 160; k++) {
  const x0 = W * (0.25 + 0.5 * alea()), y0 = H * (0.3 + 0.4 * alea()), col = couleurs[k % couleurs.length];
  const c = $("div", `position:absolute;left:${x0}px;top:${y0}px;width:${(10 + 12*alea())*u}px;height:${(16 + 14*alea())*u}px;background:${col};border-radius:${2*u}px;opacity:0`);
  tl.set(c, { opacity: 1 }, t1);
  tl.to(c, { x: (alea() - 0.5) * W * 1.2, y: H * (0.6 + 0.6 * alea()), rotation: (alea() - 0.5) * 900, rotationX: alea() * 720, duration: 2.2, ease: "power1.in" }, t1);
}
